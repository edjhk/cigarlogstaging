async function uploadPhoto(file, folder) { if (!file) { alert('uploadPhoto: no file'); return null; } alert('uploadPhoto: got file type=' + file.type + ' size=' + file.size); var blob; try { blob = await compressImage(file, 1200, 0.82); } catch (e) { alert('compressImage error: ' + e.message); return null; } if (!blob) { alert('uploadPhoto: compressImage returned null'); return null; } var path = (folder || 'feed') + '/' + Date.now() + '_' + Math.random().toString(36).slice(2) + '.jpg'; alert('uploadPhoto: uploading to ' + path); var { error } = await supa.storage .from('Photos') .upload(path, blob, { contentType: 'image/jpeg' }); if (error) { alert('uploadPhoto: upload error = ' + error.message); return null; } var url = supa.storage.from('Photos').getPublicUrl(path).data.publicUrl; alert('uploadPhoto: success, url starts with ' + url.slice(0, 60)); return url; }
